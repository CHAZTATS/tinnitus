class TinnitusTherapyGenerator {
	constructor() {
		this.audioContext = null
		this.masterGain = null
		this.pannerNode = null
		this.isPlaying = false
		this.startTime = 0
		this.timerInterval = null
		this.analyser = null
		this.animationId = null

		// Parameters from the study (Generate_Full_Experiment_Stimuli.m)
		this.sampleRate = 44100
		this.stimulusDuration = 4 // seconds
		this.rampProportion = 0.25 // 25% of duration = 1 second ramp
		this.f0Range = [96, 256] // Hz
		this.tmr = 1 // temporal modulation rate (Hz)
		this.smrRange = [1.5, 7.5] // spectral modulation rate (cycles/octave)
		this.smrCycleDuration = 8 // seconds (scyc parameter)
		this.depth = 1 // modulation depth
		this.loud = 0.1 // loudness scaling factor

		// Frequency bands: extended to support lower frequencies (experimental)
		// Standard: 1000 * 2^(0:0.5:4) = [1000, 1414, 2000, 2828, 4000, 5657, 8000, 11314, 16000]
		// Extended: adds [250, 354, 500, 707] below 1000Hz
		const fb = []
		for (let i = -4; i <= 8; i++) {
			fb.push(1000 * Math.pow(2, i * 0.5))
		}
		// 12 bands of half-octave each (4 experimental low + 8 standard)
		this.fbands = []
		for (let i = 0; i < 12; i++) {
			this.fbands.push([fb[i], fb[i + 1]])
		}
		this.standardBandOffset = 4 // first 4 bands are experimental

		// Hearing correction template per band
		// Extended with zeros for low frequencies (no correction needed below 1kHz)
		// Standard correction starts at 2kHz, reaches max at 8kHz, flat above
		this.hlCorrTemplate = [0, 0, 0, 0, 0, 0, 0, 0.125, 0.25, 0.375, 0.5, 0.5]

		// Current settings
		this.tinnitusFreq = 8000
		this.hearingCorrectionMax = 30 // dB
		this.modType = 'amp'
		this.volume = 0.3
		this.pan = 0

		this.nextScheduleTime = 0
		this.schedulerTimer = null
	}

	init() {
		if (!this.audioContext) {
			this.audioContext = new (window.AudioContext || window.webkitAudioContext)({
				sampleRate: this.sampleRate
			})
			this.masterGain = this.audioContext.createGain()
			this.pannerNode = this.audioContext.createStereoPanner()
			this.masterGain.connect(this.pannerNode)
			this.pannerNode.connect(this.audioContext.destination)
			this.analyser = this.audioContext.createAnalyser()
			this.analyser.fftSize = 256
			this.pannerNode.connect(this.analyser)
		}
		if (this.audioContext.state === 'suspended') {
			this.audioContext.resume()
		}
		this.initMediaSession()
	}

	initMediaSession() {
		if (!('mediaSession' in navigator)) return
		navigator.mediaSession.metadata = new MediaMetadata({
			title: 'Tinnitus Sound Therapy',
			artist: 'Therapy Session',
			album: 'Sound Therapy'
		})
		navigator.mediaSession.setActionHandler('play', () => { if (!this.isPlaying) togglePlay() })
		navigator.mediaSession.setActionHandler('pause', () => { if (this.isPlaying) stopPlay() })
		navigator.mediaSession.setActionHandler('stop', () => { if (this.isPlaying) stopPlay() })
	}

	// Find which band pair (2 consecutive bands = 1 octave) to modulate
	getModulationBandIndex(tinnitusFreq) {
		// Match logic from Table 1 in paper
		// Find the band pair that best centers on tinnitus frequency
		let bestIndex = this.standardBandOffset + 5 // default to 8-16kHz region
		let minDist = Infinity

		const numBands = this.fbands.length
		for (let b = 0; b < numBands - 1; b++) {
			// Band pair spans fbands[b] and fbands[b+1]
			const lowFreq = this.fbands[b][0]
			const highFreq = this.fbands[b + 1][1]
			const center = Math.sqrt(lowFreq * highFreq)
			const dist = Math.abs(Math.log2(tinnitusFreq / center))
			if (dist < minDist) {
				minDist = dist
				bestIndex = b
			}
		}
		return bestIndex
	}

	generateStimulus() {
		const duration = this.stimulusDuration
		const numSamples = Math.floor(duration * this.sampleRate)
		const rampDuration = duration * this.rampProportion

		// Random fundamental frequency
		const f0 = Math.round(this.f0Range[0] + Math.random() * (this.f0Range[1] - this.f0Range[0]))

		// Get modulation band pair index
		const modBandIndex = this.getModulationBandIndex(this.tinnitusFreq)
		// Modulation applies to bands modBandIndex and modBandIndex+1
		const modBandIndices = [modBandIndex, modBandIndex + 1]

		// Random phase offsets for modulation
		const phaseOff = Math.random() * 2 * Math.PI
		const phaseS = Math.random() * 2 * Math.PI

		// Calculate hearing correction per band (dB)
		const fbandDb = this.hlCorrTemplate.map(t => t * this.hearingCorrectionMax)

		// Collect all harmonics with their properties
		// Minimum frequency depends on whether we're using experimental low bands
		const minFreq = modBandIndices.some(i => i < this.standardBandOffset) ? 250 : 1000
		const harmonics = []
		for (let bandIdx = 0; bandIdx < this.fbands.length; bandIdx++) {
			const frange = this.fbands[bandIdx]
			// Harmonics in this band: ceil(fmin/f0) to floor((fmax-1)/f0)
			const nMin = Math.ceil(frange[0] / f0)
			const nMax = Math.floor((frange[1] - 1) / f0)

			for (let n = nMin; n <= nMax; n++) {
				const freq = n * f0
				if (freq >= minFreq && freq < 16000) {
					harmonics.push({
						n: n,
						freq: freq,
						bandIdx: bandIdx,
						isModulated: modBandIndices.includes(bandIdx),
						intensity: Math.pow(10, fbandDb[bandIdx] / 20)
					})
				}
			}
		}

		// Get modulated harmonics for calculating octave distance
		const modulatedHarmonics = harmonics.filter(h => h.isModulated)
		if (modulatedHarmonics.length === 0) {
			// Fallback if no harmonics in mod band
			return this.generateFallbackBuffer(numSamples)
		}

		const modulatedFreqs = modulatedHarmonics.map(h => h.freq)
		const minModFreq = Math.min(...modulatedFreqs)

		// Calculate log-scaled, mean-centered octave distances
		const logDistances = modulatedFreqs.map(f => Math.log2(f / minModFreq))
		const meanLogDist = logDistances.reduce((a, b) => a + b, 0) / logDistances.length
		const centeredDistances = logDistances.map(d => d - meanLogDist)

		// Create a map from freq to centered distance
		const freqToDistance = {}
		modulatedFreqs.forEach((f, i) => {
			freqToDistance[f] = centeredDistances[i]
		})

		// SMR parameters
		const smrMean = (this.smrRange[0] + this.smrRange[1]) / 2
		const smrAmplitude = (this.smrRange[1] - this.smrRange[0]) / 2

		// Generate audio buffer
		const buffer = this.audioContext.createBuffer(1, numSamples, this.sampleRate)
		const data = buffer.getChannelData(0)

		// Generate each harmonic (rand_phase = 0, so all harmonics start at sine phase)
		for (const harmonic of harmonics) {
			const freq = harmonic.freq
			const intensity = harmonic.intensity

			for (let i = 0; i < numSamples; i++) {
				const t = (i + 1) / this.sampleRate // Matlab uses 1/srate:1/srate:dur

				let sample

				if (harmonic.isModulated) {
					// Time-varying spectral modulation rate
					const smr = smrMean + smrAmplitude * Math.sin(phaseS + 2 * Math.PI * t / this.smrCycleDuration)

					// Octave distance for this harmonic
					const fDist = freqToDistance[freq]

					// Modulation phase
					const modPhase = 2 * Math.PI * (phaseOff + this.tmr * t + smr * fDist)

					if (this.modType === 'amp') {
						// Amplitude modulation: mmat = 1 + depth * sin(modPhase)
						const modValue = 1 + this.depth * Math.sin(modPhase)
						sample = modValue * Math.sin(2 * Math.PI * freq * t) * intensity
					} else {
						// Phase modulation: mmat = pi * (1 + depth * sin(modPhase)) / 2
						const phaseShift = Math.PI * (1 + this.depth * Math.sin(modPhase)) / 2
						sample = Math.sin(2 * Math.PI * freq * t + phaseShift) * intensity
					}
				} else {
					// No modulation
					sample = Math.sin(2 * Math.PI * freq * t) * intensity
				}

				data[i] += sample
			}
		}

		// Calculate RMS and scale
		let sumSq = 0
		for (let i = 0; i < numSamples; i++) {
			sumSq += data[i] * data[i]
		}
		const rms = Math.sqrt(sumSq / numSamples)
		const scale = this.loud / (10 * rms)

		for (let i = 0; i < numSamples; i++) {
			data[i] *= scale
		}

		// Apply raised cosine ramp (wind_ramp function)
		const rampSamples = Math.round(2 * rampDuration * this.sampleRate)
		const halfRamp = Math.round(rampSamples / 2)

		for (let i = 0; i < halfRamp; i++) {
			// w = linspace(-pi/2, 1.5*pi, wds) -> w = (sin(w)+1)/2
			const w = -Math.PI / 2 + (2 * Math.PI) * (i / (rampSamples - 1))
			const rampValue = (Math.sin(w) + 1) / 2
			data[i] *= rampValue
		}

		for (let i = 0; i < halfRamp; i++) {
			const idx = numSamples - halfRamp + i
			const wIdx = halfRamp + i
			const w = -Math.PI / 2 + (2 * Math.PI) * (wIdx / (rampSamples - 1))
			const rampValue = (Math.sin(w) + 1) / 2
			if (idx < numSamples) {
				data[idx] *= rampValue
			}
		}

		// Prevent clipping
		let maxAbs = 0
		for (let i = 0; i < numSamples; i++) {
			maxAbs = Math.max(maxAbs, Math.abs(data[i]))
		}
		if (maxAbs > 1) {
			for (let i = 0; i < numSamples; i++) {
				data[i] /= maxAbs
			}
		}

		return buffer
	}

	generateFallbackBuffer(numSamples) {
		const buffer = this.audioContext.createBuffer(1, numSamples, this.sampleRate)
		const data = buffer.getChannelData(0)
		// Generate simple noise as fallback
		for (let i = 0; i < numSamples; i++) {
			data[i] = (Math.random() * 2 - 1) * 0.1
		}
		return buffer
	}

	scheduleNextStimulus() {
		if (!this.isPlaying) return

		const buffer = this.generateStimulus()
		const source = this.audioContext.createBufferSource()
		source.buffer = buffer
		source.connect(this.masterGain)

		const currentTime = this.audioContext.currentTime
		const startTime = Math.max(currentTime, this.nextScheduleTime)

		source.start(startTime)
		this.nextScheduleTime = startTime + this.stimulusDuration

		// Schedule next stimulus before current one ends
		const scheduleAhead = (this.stimulusDuration - 0.5) * 1000
		this.schedulerTimer = setTimeout(() => this.scheduleNextStimulus(), scheduleAhead)
	}

	start() {
		this.init()
		this.isPlaying = true
		this.startTime = Date.now()
		this.nextScheduleTime = this.audioContext.currentTime

		this.masterGain.gain.setValueAtTime(this.volume, this.audioContext.currentTime)
		this.pannerNode.pan.setValueAtTime(this.pan, this.audioContext.currentTime)

		this.scheduleNextStimulus()
		this.startTimer()
		this.startVisualizer()
	}

	stop() {
		this.isPlaying = false

		if (this.schedulerTimer) {
			clearTimeout(this.schedulerTimer)
			this.schedulerTimer = null
		}

		if (this.masterGain) {
			this.masterGain.gain.setValueAtTime(0, this.audioContext.currentTime)
		}

		this.stopTimer()
		this.stopVisualizer()
	}

	setVolume(value) {
		this.volume = value
		if (this.masterGain && this.isPlaying) {
			this.masterGain.gain.setValueAtTime(value, this.audioContext.currentTime)
		}
	}

	setPan(value) {
		this.pan = value
		if (this.pannerNode) {
			this.pannerNode.pan.setValueAtTime(value, this.audioContext.currentTime)
		}
	}

	startTimer() {
		this.timerInterval = setInterval(() => {
			const elapsed = Math.floor((Date.now() - this.startTime) / 1000)
			const hours = Math.floor(elapsed / 3600)
			const minutes = Math.floor((elapsed % 3600) / 60)
			const seconds = elapsed % 60
			document.getElementById('timer').textContent =
				`${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
		}, 1000)
	}

	stopTimer() {
		if (this.timerInterval) {
			clearInterval(this.timerInterval)
			this.timerInterval = null
		}
	}

	startVisualizer() {
		const canvas = document.getElementById('visualizer')
		const ctx = canvas.getContext('2d')
		const bufferLength = this.analyser.frequencyBinCount
		const dataArray = new Uint8Array(bufferLength)

		const draw = () => {
			if (!this.isPlaying) return
			this.animationId = requestAnimationFrame(draw)
			this.analyser.getByteFrequencyData(dataArray)

			ctx.fillStyle = '#0f0f23'
			ctx.fillRect(0, 0, canvas.width, canvas.height)

			const barWidth = (canvas.width / bufferLength) * 2.5
			let x = 0

			for (let i = 0; i < bufferLength; i++) {
				const barHeight = (dataArray[i] / 255) * canvas.height
				const hue = 180 + (i / bufferLength) * 60
				ctx.fillStyle = `hsl(${hue}, 80%, 50%)`
				ctx.fillRect(x, canvas.height - barHeight, barWidth, barHeight)
				x += barWidth + 1
			}
		}
		draw()
	}

	stopVisualizer() {
		if (this.animationId) {
			cancelAnimationFrame(this.animationId)
			this.animationId = null
		}
		const canvas = document.getElementById('visualizer')
		const ctx = canvas.getContext('2d')
		ctx.fillStyle = '#0f0f23'
		ctx.fillRect(0, 0, canvas.width, canvas.height)
	}

	previewTone(freq, duration = 0.5) {
		this.init()
		const numSamples = Math.floor(duration * this.sampleRate)
		const buffer = this.audioContext.createBuffer(1, numSamples, this.sampleRate)
		const data = buffer.getChannelData(0)

		for (let i = 0; i < numSamples; i++) {
			const t = i / this.sampleRate
			data[i] = 0.2 * Math.sin(2 * Math.PI * freq * t)
		}

		const rampSamples = Math.floor(0.05 * this.sampleRate)
		for (let i = 0; i < rampSamples; i++) {
			const rampValue = 0.5 * (1 - Math.cos(Math.PI * i / rampSamples))
			data[i] *= rampValue
			data[numSamples - 1 - i] *= rampValue
		}

		const source = this.audioContext.createBufferSource()
		source.buffer = buffer
		const gain = this.audioContext.createGain()
		gain.gain.setValueAtTime(this.volume, this.audioContext.currentTime)
		source.connect(gain)
		gain.connect(this.audioContext.destination)
		source.start()
	}
}

class SessionTracker {
	storageKey = 'tinnitus-sessions'
	currentSessionStart = null

	getSessions() {
		const data = localStorage.getItem(this.storageKey)
		return data ? JSON.parse(data) : []
	}

	saveSessions(sessions) {
		localStorage.setItem(this.storageKey, JSON.stringify(sessions))
	}

	startSession() {
		this.currentSessionStart = Date.now()
	}

	endSession() {
		if (!this.currentSessionStart) return

		const end = Date.now()
		const durationMs = end - this.currentSessionStart
		const durationMinutes = durationMs / 60000

		if (durationMinutes >= 1) {
			const sessions = this.getSessions()
			sessions.push({
				start: this.currentSessionStart,
				end: end,
				duration: durationMs
			})
			this.saveSessions(sessions)
		}

		this.currentSessionStart = null
		this.renderHistory()
	}

	getSessionsByDay() {
		const sessions = this.getSessions()
		const byDay = {}

		for (const session of sessions) {
			const date = new Date(session.start).toLocaleDateString('fr-FR', {
				weekday: 'short',
				day: 'numeric',
				month: 'short'
			})
			if (!byDay[date]) byDay[date] = 0
			byDay[date] += session.duration
		}

		return byDay
	}

	formatDuration(ms) {
		const totalSeconds = Math.floor(ms / 1000)
		const hours = Math.floor(totalSeconds / 3600)
		const minutes = Math.floor((totalSeconds % 3600) / 60)
		if (hours > 0) return `${hours}h ${minutes}min`
		return `${minutes} min`
	}

	renderHistory() {
		const container = document.getElementById('historyContent')
		const byDay = this.getSessionsByDay()
		const days = Object.keys(byDay)

		if (days.length === 0) {
			container.textContent = 'No sessions recorded yet.'
			return
		}

		const rows = days.map(day =>
			`<div style="display:flex;justify-content:space-between;padding:8px 0;border-bottom:1px solid #333;">
				<span>${day}</span>
				<span style="color:#00d4ff;font-weight:bold;">${this.formatDuration(byDay[day])}</span>
			</div>`
		).join('')

		const totalMs = Object.values(byDay).reduce((a, b) => a + b, 0)
		container.innerHTML = rows +
			`<div style="display:flex;justify-content:space-between;padding:12px 0;margin-top:8px;font-weight:bold;">
				<span>Total (${days.length} days)</span>
				<span style="color:#00d4ff;">${this.formatDuration(totalMs)}</span>
			</div>`
	}
}

// Settings persistence
const SETTINGS_KEY = 'tinnitus-settings'

function loadSettings() {
	const data = localStorage.getItem(SETTINGS_KEY)
	return data ? JSON.parse(data) : { freqIndex: 12, volume: 30, hearingProfile: '30', modType: 'amp', pan: 0 }
}

function saveSettings(settings) {
	localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings))
}

function updateSetting(key, value) {
	const settings = loadSettings()
	settings[key] = value
	saveSettings(settings)
}

const generator = new TinnitusTherapyGenerator()
const sessionTracker = new SessionTracker()
sessionTracker.renderHistory()

// Load saved settings
const savedSettings = loadSettings()

// Frequency options from the study (17 frequencies in 1/4 octave steps)
const frequencies = [1000, 1200, 1400, 1700, 2000, 2400, 2800, 3400, 4000, 4800, 5700, 6700, 8000, 9500, 11000, 13000, 16000]
let selectedFreqIndex = savedSettings.freqIndex

// Custom frequency override via URL parameter (?freq=580)
const urlParams = new URLSearchParams(window.location.search)
const customFreqParam = urlParams.get('freq')
let customFreq = null
if (customFreqParam) {
	const parsed = parseInt(customFreqParam)
	if (parsed >= 100 && parsed <= 20000) {
		customFreq = parsed
		generator.tinnitusFreq = customFreq
	}
}

// Create frequency buttons
const freqButtonsContainer = document.getElementById('freqButtons')
frequencies.forEach((freq, index) => {
	const btn = document.createElement('button')
	btn.className = 'freq-btn' + (index === selectedFreqIndex ? ' selected' : '')
	btn.textContent = freq >= 1000 ? (freq / 1000).toFixed(1).replace('.0', '') + ' kHz' : freq + ' Hz'
	btn.onclick = () => selectFrequency(index)
	freqButtonsContainer.appendChild(btn)
})

function formatFreq(freq) {
	return freq >= 1000 ? (freq / 1000).toFixed(1).replace('.0', '') + ' kHz' : freq + ' Hz'
}

function selectFrequency(index, preview = true) {
	selectedFreqIndex = index
	if (!customFreq) {
		generator.tinnitusFreq = frequencies[index]
	}
	updateSetting('freqIndex', index)

	document.querySelectorAll('.freq-btn').forEach((btn, i) => {
		btn.classList.toggle('selected', i === index)
	})

	updateFrequencyDisplay()
	if (preview && !customFreq) generator.previewTone(frequencies[index])
}

function updateFrequencyDisplay() {
	const displayEl = document.getElementById('selectedFreq')
	if (customFreq) {
		const isExperimental = customFreq < 1000
		displayEl.innerHTML = `<span style="color:#ff9800;">${formatFreq(customFreq)} (URL override${isExperimental ? ' - experimental' : ''})</span>`
	} else {
		displayEl.textContent = formatFreq(frequencies[selectedFreqIndex])
	}
}

// Play controls
function togglePlay() {
	const playBtn = document.getElementById('playBtn')
	const stopBtn = document.getElementById('stopBtn')
	const status = document.getElementById('status')
	const statusText = document.getElementById('statusText')

	generator.hearingCorrectionMax = parseInt(document.getElementById('hearingProfile').value)
	generator.modType = document.getElementById('modType').value
	generator.setVolume(volumeSlider.value / 100)
	generator.setPan(panSlider.value / 100)

	generator.start()
	sessionTracker.startSession()

	playBtn.disabled = true
	stopBtn.disabled = false
	status.className = 'status playing'
	statusText.textContent = 'Therapy in progress...'
}

function stopPlay() {
	const playBtn = document.getElementById('playBtn')
	const stopBtn = document.getElementById('stopBtn')
	const status = document.getElementById('status')
	const statusText = document.getElementById('statusText')

	generator.stop()
	sessionTracker.endSession()

	playBtn.disabled = false
	stopBtn.disabled = true
	status.className = 'status stopped'
	statusText.textContent = 'Stopped'
}
